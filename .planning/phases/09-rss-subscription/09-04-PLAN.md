    ```python
    import asyncio
    import json
    import click
    import typer
    from typing import Optional
    from rich.console import Console
    from rich.table import Table

    from saw.db.feed_models import Feed, FeedEntry
    from saw.engines.ingest.feed_manager import FeedManager, FeedManagerError
    from saw.config.settings import get_db_session

    app = typer.Typer(name="feed", help="RSS feed subscription management")
    console = Console()

    @app.command("add")
    def add_feed(
        url: str = typer.Argument(..., help="RSS/Atom feed URL"),
        category: Optional[str] = typer.Option(None, "--category", "-c", help="Feed category"),
        tags: Optional[str] = typer.Option(None, "--tags", "-t", help="Comma-separated filter keywords"),
        interval: int = typer.Option(3600, "--interval", "-i", help="Poll interval in seconds"),
    ):
        """Subscribe to an RSS/Atom feed.

        Per RSSS-01: Subscribe to RSS/Atom Feed.
        Per RSSS-04: Configure sync frequency.
        Per RSSS-07: Filter by keywords.
        """
        async def _add():
            async with get_db_session() as db:
                manager = FeedManager(db)
                tag_list = [t.strip() for t in tags.split(",")] if tags else None

                try:
                    feed_id = await manager.add_feed(
                        url=url,
                        category=category,
                        tags=tag_list,
                        poll_interval=interval,
                    )
                    console.print(f"[green]Subscribed to feed:[/green] {feed_id}")
                    console.print(f"  URL: {url}")
                    if category:
                        console.print(f"  Category: {category}")
                except FeedManagerError as e:
                    console.print(f"[red]Error:[/red] {e}")
                    raise typer.Exit(1)

        asyncio.run(_add())

    @app.command("list")
    def list_feeds(
        category: Optional[str] = typer.Option(None, "--category", "-c", help="Filter by category"),
        all: bool = typer.Option(False, "--all", "-a", help="Show inactive feeds too"),
    ):
        """List all feed subscriptions.

        Per RSSS-06: Feed classification management.
        """
        async def _list():
            async with get_db_session() as db:
                query = db.query(Feed)
                if not all:
                    query = query.filter(Feed.active == True)
                if category:
                    query = query.filter(Feed.category == category)

                feeds = query.all()

                if not feeds:
                    console.print("[yellow]No feeds found[/yellow]")
                    return

                table = Table(title="Feed Subscriptions")
                table.add_column("ID", style="cyan", no_wrap=True)
                table.add_column("Title", style="green")
                table.add_column("URL", style="blue")
                table.add_column("Category", style="magenta")
                table.add_column("Interval", justify="right")
                table.add_column("Status", style="yellow")

                for feed in feeds:
                    status = "[green]Active[/green]" if feed.active else "[red]Inactive[/red]"
                    table.add_row(
                        feed.id[:8],
                        feed.title or feed.url[:30],
                        feed.url[:40] + "..." if len(feed.url) > 40 else feed.url,
                        feed.category or "-",
                        f"{feed.poll_interval}s",
                        status,
                    )

                console.print(table)

        asyncio.run(_list())

    @app.command("poll")
    def poll_feed(
        feed_id: str = typer.Argument(..., help="Feed ID to poll"),
    ):
        """Manually poll a feed for new entries."""
        async def _poll():
            async with get_db_session() as db:
                manager = FeedManager(db)
                try:
                    result = await manager.poll_feed(feed_id)
                    console.print(f"[green]Poll complete for feed:[/green] {feed_id}")
                    console.print(f"  New entries: {result.new_entries}")
                    console.print(f"  Updated entries: {result.updated_entries}")
                    console.print(f"  Skipped: {result.skipped_entries}")
                    if result.not_modified:
                        console.print("  [yellow]Feed not modified (304)[/yellow]")
                    if result.errors:
                        for err in result.errors:
                            console.print(f"  [red]Error:[/red] {err}")
                except FeedManagerError as e:
                    console.print(f"[red]Error:[/red] {e}")
                    raise typer.Exit(1)

        asyncio.run(_poll())

    @app.command("remove")
    def remove_feed(
        feed_id: str = typer.Argument(..., help="Feed ID to remove"),
    ):
        """Unsubscribe from a feed (soft delete)."""
        async def _remove():
            async with get_db_session() as db:
                feed = db.query(Feed).filter(Feed.id == feed_id).first()
                if not feed:
                    console.print(f"[red]Feed not found:[/red] {feed_id}")

                feed.active = False
                db.commit()
                console.print(f"[green]Unsubscribed from feed:[/green] {feed_id}")

        asyncio.run(_remove())

    @app.command("update")
    def update_feed(
        feed_id: str = typer.Argument(..., help="Feed ID to update"),
        category: Optional[str] = typer.Option(None, "--category", "-c", help="New category"),

        tags: Optional[str] = typer.Option(None, "--tags", "-t", help="New filter keywords"),

        """Update feed settings."""
        async def _update():
            async with get_db_session() as db:
                feed = db.query(Feed).filter(Feed.id == feed_id).first()
                if not feed:
                    console.print(f"[red]Feed not found:[/red] {feed_id}")
                    raise typer.Exit(1)

                if category is not None:
                    feed.category = category
                if interval is not None:
                    feed.poll_interval = interval
                if tags is not None:

    @app.command("entries")
    def list_entries(

        """List entries for a feed."""
        async def _entries():

                    query = query.filter(FeedEntry.status == status)

                entries = query.order_by(FeedEntry.first_seen_at.desc()).limit(limit).all()

                if not entries:
                    console.print("[yellow]No entries found[/yellow]")

                table = Table(title=f"Entries for Feed {feed_id[:8]}")
                table.add_column("Title", style="green")
                table.add_column("Status", style="yellow")
                table.add_column("First Seen", style="blue")

                for entry in entries:
                    table.add_row(
                        entry.title[:50] + "..." if len(entry.title) > 50 else entry.title,
                        entry.status,
                        entry.first_seen_at.strftime("%Y-%m-%d %H:%M") if entry.first_seen_at else "-",
                    )

                console.print(table)

        asyncio.run(_entries())
    ```

    ```python
    from __future__ import annotations

    import asyncio
    import logging
    from dataclasses import dataclass
    from datetime import datetime, timedelta
    from typing import Optional

    from apscheduler.schedulers.asyncio import AsyncIOScheduler
    from apscheduler.triggers.interval import IntervalTrigger
    from apscheduler.jobstores.memory import MemoryJobStore

    from saw.db.feed_models import Feed
    from saw.engines.ingest.feed_manager import FeedManager, FeedManagerError, PollResult

    logger = logging.getLogger(__name__)

    @dataclass
    class FeedJobConfig:
        """Configuration for a feed polling job."""
        feed_id: str
        feed_url: str
        poll_interval: int
        consecutive_failures: int = 0
        last_failure: Optional[datetime] = None

    class FeedScheduler:
        """APScheduler-based feed polling scheduler.

        Per Pitfall 27: Implements adaptive intervals and exponential backoff.
        Per RSSS-04: Configurable sync frequency.
        """

        def __init__(self, feed_manager: FeedManager, db_session):
            self._manager = feed_manager
            self._db = db_session
            self._scheduler = AsyncIOScheduler(
                jobstores={"default": MemoryJobStore()},
            )
            self._job_configs: dict[str, FeedJobConfig] = {}
            self._running = False

        async def start(self):
            """Start the scheduler and load all active feeds."""
            if self._running:
                logger.warning("Scheduler already running")
                return

            self._scheduler.start()
            self._running = True

            # Load all active feeds from database
            feeds = self._db.query(Feed).filter(Feed.active == True).all()
            for feed in feeds:
                await self.add_feed_job(feed)

            logger.info(f"Scheduler started with {len(feeds)} feeds")

        async def stop(self):
            """Stop the scheduler."""
            if not self._running:
                return

            self._scheduler.shutdown(wait=True)
            self._running = False
            logger.info("Scheduler stopped")

        async def add_feed_job(self, feed: Feed):
            """Schedule a polling job for a feed.

            Per Pitfall 27: Uses adaptive interval calculation.
            """
            job_id = f"feed_poll_{feed.id}"

            # Calculate adaptive interval
            adaptive_interval = self._manager.calculate_adaptive_interval(feed.id)

            # Use the lower of adaptive or configured interval
            interval = min(adaptive_interval, feed.poll_interval)

            config = FeedJobConfig(
                feed_id=feed.id,
                feed_url=feed.url,
                poll_interval=interval,
            )

            self._job_configs[feed.id] = config

            self._scheduler.add_job(
                self._poll_feed_job,
                trigger=IntervalTrigger(seconds=interval),
                id=job_id,
                args=[feed.id],
                name=f"Poll {feed.title or feed.url}",
                misfire_grace_time=300,  # 5 minute grace period
                max_instances=1,  # Only one instance per feed at a time
            )

            logger.info(f"Scheduled feed {feed.id} with interval {interval}s")

        async def remove_feed_job(self, feed_id: str):
            """Remove a feed's polling job."""
            job_id = f"feed_poll_{feed_id}"

            if self._scheduler.get_job(job_id):
                self._scheduler.remove_job(job_id)
                logger.info(f"Removed job for feed {feed_id}")

            if feed_id in self._job_configs:
                del self._job_configs[feed_id]

        async def update_feed_interval(self, feed_id: str, new_interval: int):
            """Update a feed's polling interval."""
            job_id = f"feed_poll_{feed_id}"

            job = self._scheduler.get_job(job_id)
            if not job:
                logger.warning(f"Job not found for feed {feed_id}")
                return

            # Reschedule with new interval
            self._scheduler.reschedule_job(
                job_id,
                trigger=IntervalTrigger(seconds=new_interval),
            )

            if feed_id in self._job_configs:
                self._job_configs[feed_id].poll_interval = new_interval

            logger.info(f"Updated interval for feed {feed_id} to {new_interval}s")

        async def _poll_feed_job(self, feed_id: str):
            """Execute a feed poll job.

            Per Pitfall 27: Implements exponential backoff for failures.
            """
            config = self._job_configs.get(feed_id)
            if not config:
                logger.warning(f"No config for feed {feed_id}")
                return

            try:
                result = await self._manager.poll_feed(feed_id)

                # Reset failure count on success
                if result.errors and "HTTP" not in result.errors[0]:
                    # Non-HTTP error, still count as failure
                    pass
                else:
                    config.consecutive_failures = 0

                logger.info(
                    f"Poll complete for {feed_id}: "
                    f"{result.new_entries} new, {result.updated_entries} updated"
                )

            except FeedManagerError as e:
                logger.error(f"Poll failed for {feed_id}: {e}")

                # Exponential backoff
                config.consecutive_failures += 1
                config.last_failure = datetime.utcnow()

                backoff_interval = min(
                    config.poll_interval * (2 ** config.consecutive_failures),
                    86400,  # Max 24 hours
                )

                # Reschedule with backoff
                await self.update_feed_interval(feed_id, backoff_interval)

                logger.warning(
                    f"Backing off feed {feed_id}: "
                    f"{config.consecutive_failures} failures, "
                    f"next poll in {backoff_interval}s"
                )

            except Exception as e:
                logger.exception(f"Unexpected error polling {feed_id}: {e}")

    # Module-level functions for CLI integration
    _scheduler_instance: Optional[FeedScheduler] = None

    async def start_scheduler(feed_manager: FeedManager, db_session) -> FeedScheduler:
        """Start the global feed scheduler."""
        global _scheduler_instance

        if _scheduler_instance is not None:
            return _scheduler_instance

        _scheduler_instance = FeedScheduler(feed_manager, db_session)
        await _scheduler_instance.start()
        return _scheduler_instance

    async def stop_scheduler():
        """Stop the global feed scheduler."""
        global _scheduler_instance

        if _scheduler_instance is not None:
            await _scheduler_instance.stop()
            _scheduler_instance = None
    ```

    ```python
    import random

    class FeedScheduler:
        # ... existing code ...

        async def start(self):
            """Start the scheduler with staggered feed polling.

            Per Pitfall 27: Stagger polling across time windows.
            """
            if self._running:
                logger.warning("Scheduler already running")
                return

            self._scheduler.start()
            self._running = True

            # Load all active feeds
            feeds = self._db.query(Feed).filter(Feed.active == True).all()

            # Group feeds by poll_interval
            interval_groups: dict[int, list[Feed]] = {}
            for feed in feeds:
                if feed.poll_interval not in interval_groups:
                    interval_groups[feed.poll_interval] = []
                interval_groups[feed.poll_interval].append(feed)

            # Stagger feeds within each interval group
            for interval, group_feeds in interval_groups.items():
                stagger_delay = interval / len(group_feeds)

                for i, feed in enumerate(group_feeds):
                    # Calculate initial delay (staggered start)
                    initial_delay = int(stagger_delay * i)
                    # Add small random jitter to avoid exact alignment
                    jitter = random.randint(0, min(60, stagger_delay // 2))

                    await self._add_staggered_feed_job(
                        feed,
                        initial_delay=initial_delay + jitter,
                    )

            logger.info(f"Scheduler started with {len(feeds)} feeds (staggered)")

        async def _add_staggered_feed_job(
            self,
            feed: Feed,
            initial_delay: int = 0,
        ):
            """Add a feed job with staggered start time."""
            job_id = f"feed_poll_{feed.id}"

            adaptive_interval = self._manager.calculate_adaptive_interval(feed.id)
            interval = min(adaptive_interval, feed.poll_interval)

            config = FeedJobConfig(
                feed_id=feed.id,
                feed_url=feed.url,
                poll_interval=interval,
            )
            self._job_configs[feed.id] = config

            # Calculate next run time
            next_run = datetime.utcnow() + timedelta(seconds=initial_delay)

            self._scheduler.add_job(
                self._poll_feed_job,
                trigger=IntervalTrigger(
                    seconds=interval,
                    start_date=next_run,
                ),
                id=job_id,
                args=[feed.id],
                name=f"Poll {feed.title or feed.url}",
                next_run_time=next_run,
                misfire_grace_time=300,
                max_instances=1,
            )

            logger.info(
                f"Scheduled feed {feed.id} with interval {interval}s, "
                f"first poll in {initial_delay}s"
            )

        def _calculate_stagger_delay(self, interval: int, feed_count: int) -> int:
            """Calculate stagger delay for a feed.

            Per Pitfall 27: Distribute polls evenly across interval.
            """
            if feed_count <= 1:
                return 0

            base_delay = interval // feed_count
            jitter = random.randint(0, min(60, base_delay // 2))

            return base_delay + jitter
    ```

    ```python
    from fastapi import FastAPI
    from saw.engines.ingest.scheduler import start_scheduler, stop_scheduler

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        """Application lifespan handler for scheduler management."""
        # Startup
        async with get_db_session() as db:
            feed_manager = FeedManager(db)
            scheduler = await start_scheduler(feed_manager, db)
            app.state.scheduler = scheduler

        yield  # Application runs

        # Shutdown
        await stop_scheduler()

    # Create app with lifespan
    app = FastAPI(
        title="Smart Agent Wiki",
        lifespan=lifespan,
    )
    ```
