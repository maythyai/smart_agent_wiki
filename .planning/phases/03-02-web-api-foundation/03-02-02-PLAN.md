```python
# src/saw/drivers/web/routes/search.py
from fastapi import APIRouter, Depends, Request, Query, HTTPException
from typing import Optional

from saw.drivers.web.schemas.search import SearchQuery, SearchResult, SearchResponse

router = APIRouter()

def get_query_engine(request: Request):
    """Dependency: get QueryEngine from app.state."""
    return request.app.state.query

@router.get("/search", response_model=SearchResponse)
async def search(
    q: str = Query(..., min_length=1, description="Search query"),
    page: int = Query(1, ge=1, description="Page number"),
    per_page: int = Query(10, ge=1, le=100, description="Results per page"),
    type: Optional[str] = Query(None, description="Filter by page type"),
    tag: Optional[str] = Query(None, description="Filter by tag"),
    min_confidence: Optional[int] = Query(None, ge=1, le=4, description="Min confidence level"),
    engine = Depends(get_query_engine),
) -> SearchResponse:
    """Search knowledge base using BM25 + FTS5 (per D-07).
    
    Per D-08: Results include snippet, citation, confidence.
    Per D-09: Support pagination and filtering.
    """
    # Execute search via QueryEngine
    result = engine.query(question=q, mode="search")
    
    # Convert QueryResult to SearchResponse
    results = []
    for i, source in enumerate(result.sources):
        # Apply confidence filter
        confidence_value = source.get("confidence", 1)
        if isinstance(confidence_value, str):
            # Map string to int
            conf_map = {"unverified": 1, "single_source": 2, "cross_validated": 3, "human_verified": 4}
            confidence_int = conf_map.get(confidence_value.lower(), 1)
        else:
            confidence_int = confidence_value
        
        if min_confidence and confidence_int < min_confidence:
            continue
        
        # Apply type filter
        if type and source.get("type") != type:
            continue
        
        # Apply tag filter (if source has tags)
        if tag and tag not in source.get("tags", []):
            continue
        
        # Create snippet
        content = source.get("content", "")
        snippet = content[:200] + "..." if len(content) > 200 else content
        
        results.append(SearchResult(
            slug=source.get("slug", source.get("claim_uuid", "")),
            title=source.get("title", source.get("claim_uuid", "")),
            snippet=snippet,
            confidence=confidence_int,
            freshness=source.get("freshness", 0),
            citations=[f"claim:{source.get('claim_uuid', '')}"],
            score=source.get("score", 0.0),
        ))
    
    # Pagination
    total = len(results)
    offset = (page - 1) * per_page
    paginated = results[offset:offset + per_page]
    
    return SearchResponse(
        results=paginated,
        total=total,
        page=page,
        per_page=per_page,
        has_more=offset + per_page < total,
    )

@router.get("/search/suggestions")
async def search_suggestions(
    q: str = Query(..., min_length=1, description="Partial query"),
    limit: int = Query(5, ge=1, le=20, description="Max suggestions"),
    engine = Depends(get_query_engine),
) -> list[str]:
    """Get search suggestions based on partial query."""
    # Use FTS5 prefix search
    result = engine.query(question=q, mode="search")
    
    # Extract unique titles
    titles = []
    seen = set()
    for source in result.sources[:limit * 2]:
        title = source.get("title", "")
        if title and title not in seen:
            titles.append(title)
            seen.add(title)
        if len(titles) >= limit:
            break
    
    return titles
```

```python
from pydantic import BaseModel, Field
from typing import Optional
from enum import Enum

class TraversalMode(str, Enum):
    bfs = "bfs"
    dfs = "dfs"

class GraphQuery(BaseModel):
    """Graph query parameters (per D-10~12)."""
    depth: int = Field(2, ge=1, le=5, description="Traversal depth")
    mode: TraversalMode = Field(TraversalMode.bfs, description="BFS or DFS")
    type: Optional[str] = Field(None, description="Filter by entity type")
    max_nodes: int = Field(50, ge=1, le=200, description="Max nodes to return")

class GraphNode(BaseModel):
    """Node in knowledge graph."""
    id: str
    label: str
    type: str
    confidence: int = Field(1, ge=1, le=4)
    description: Optional[str] = None

class GraphEdge(BaseModel):
    """Edge in knowledge graph."""
    id: str
    source: str
    target: str
    type: str
    weight: float = Field(1.0, ge=0.0)

class GraphResponse(BaseModel):
    """Graph data for Cytoscape.js."""
    nodes: list[GraphNode]
    edges: list[GraphEdge]
    total_nodes: int
    total_edges: int
```

```python
# src/saw/drivers/web/routes/graph.py
from fastapi import APIRouter, Depends, Request, Query, Path, HTTPException
from typing import Optional

from saw.drivers.web.schemas.graph import (
    TraversalMode, GraphQuery, GraphNode, GraphEdge, GraphResponse
)

router = APIRouter()

def get_query_engine(request: Request):
    """Dependency: get QueryEngine from app.state."""
    return request.app.state.query

@router.get("/graph", response_model=GraphResponse)
async def get_graph(
    depth: int = Query(2, ge=1, le=5, description="Traversal depth"),
    mode: TraversalMode = Query(TraversalMode.bfs, description="BFS or DFS"),
    type: Optional[str] = Query(None, description="Filter by entity type"),
    max_nodes: int = Query(50, ge=1, le=200, description="Max nodes"),
    engine = Depends(get_query_engine),
) -> GraphResponse:
    """Get knowledge graph nodes and edges (per D-10).
    
    Per D-12: Support BFS/DFS traversal parameters.
    """
    # Use GraphTraverse to get all nodes
    graph = engine._graph
    
    # Get all entities (no specific start node)
    all_entities = list(graph._entity_cache.values())
    
    # Apply type filter if specified
    if type:
        all_entities = [e for e in all_entities if e.entity_type == type]
    
    # Limit nodes
    entities = all_entities[:max_nodes]
    
    # Build nodes
    nodes = []
    for entity in entities:
        nodes.append(GraphNode(
            id=entity.uuid,
            label=entity.name,
            type=entity.entity_type,
            confidence=1,  # Default confidence
            description=entity.description[:100] if entity.description else None,
        ))
    
    # Build edges from entities
    edges = []
    entity_ids = {e.uuid for e in entities}
    edge_id = 0
    
    # Get edges from NetworkX graph
    for source_uuid, target_uuid, data in graph._graph.edges(data=True):
        if source_uuid in entity_ids and target_uuid in entity_ids:
            edges.append(GraphEdge(
                id=f"edge-{edge_id}",
                source=source_uuid,
                target=target_uuid,
                type=data.get("relation_type", "related_to"),
                weight=data.get("weight", 1.0),
            ))
            edge_id += 1
    
    return GraphResponse(
        nodes=nodes,
        edges=edges,
        total_nodes=len(nodes),
        total_edges=len(edges),
    )

@router.get("/graph/{entity}", response_model=GraphResponse)
async def get_entity_subgraph(
    entity: str = Path(..., description="Entity ID or name"),
    depth: int = Query(2, ge=1, le=5, description="Traversal depth"),
    mode: TraversalMode = Query(TraversalMode.bfs, description="BFS or DFS"),
    max_nodes: int = Query(50, ge=1, le=200, description="Max nodes"),
    engine = Depends(get_query_engine),
) -> GraphResponse:
    """Get entity details and relationships (per D-11~12).
    
    Per D-12: Support BFS/DFS traversal with depth parameter.
    """
    graph = engine._graph
    
    # Traverse from entity
    result = graph.traverse(
        entity_name=entity,
        mode=mode.value,
        max_depth=depth,
        max_nodes=max_nodes,
    )
    
    if not result.nodes:
        raise HTTPException(status_code=404, detail=f"Entity '{entity}' not found")
    
    # Build nodes
    nodes = []
    for node in result.nodes:
        nodes.append(GraphNode(
            id=node.uuid,
            label=node.name,
            type=node.entity_type,
            confidence=1,
            description=node.description[:100] if node.description else None,
        ))
    
    # Build edges
    edges = []
    for i, edge in enumerate(result.edges):
        edges.append(GraphEdge(
            id=f"edge-{i}",
            source=edge.source_uuid,
            target=edge.target_uuid,
            type=edge.relation_type,
            weight=edge.weight,
        ))
    
    return GraphResponse(
        nodes=nodes,
        edges=edges,
        total_nodes=len(nodes),
        total_edges=len(edges),
    )
```
