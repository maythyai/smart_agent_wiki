### MCP Tool Implementation

```python
# saw/mcp/tools/impact.py
"""MCP tool for impact analysis."""
from mcp.server import Server
from saw.analysis.impact import analyze_impact, NodeNotFoundError
import logging

logger = logging.getLogger(__name__)


def register_impact_tool(server: Server):
    """Register saw_impact tool with MCP server."""

    @server.tool()
    async def saw_impact(
        target: str,
        direction: str = 'upstream',
        max_depth: int = 3,
        min_confidence: float = 0.8,
        relation_types: list[str] = None,
        include_tests: bool = False
    ) -> dict:
        """
        Analyze code modification impact.

        Identifies what will be affected if you modify the target symbol.
        Use this BEFORE making changes to understand blast radius.

        Args:
            target: Symbol name or UID to analyze (e.g., 'UserService', 'handleLogin')
            direction: 'upstream' (what depends on this) or 'downstream' (what this depends on)
            max_depth: Maximum traversal depth 1-5 (default: 3)
            min_confidence: Minimum edge confidence 0.0-1.0 (default: 0.8)
            relation_types: Filter relations ['CALLS', 'IMPORTS', 'EXTENDS', 'IMPLEMENTS']
            include_tests: Include test files in results (default: false)

        Returns:
            ImpactResult with affected nodes grouped by depth and risk level

        Example:
            saw_impact(target="UserService", direction="upstream")
            # Returns: {impacts: [...], summary: {high_risk_count: 5, ...}}
        """
        graph = get_current_graph()  # From context

        try:
            result = analyze_impact(
                graph, target, direction, max_depth,
                min_confidence, relation_types, include_tests
            )

            # Warning for high-risk modifications
            if result['summary']['high_risk_count'] > 0:
                logger.warning(
                    f"HIGH RISK: Modifying {target} will break "
                    f"{result['summary']['high_risk_count']} direct dependents"
                )

            return result

        except NodeNotFoundError as e:
            return {
                'error': 'node_not_found',
                'message': str(e),
                'suggestions': _find_similar_nodes(graph, target)
            }


def _find_similar_nodes(graph, target: str) -> list[str]:
    """Find similar node names for suggestions."""
    all_nodes = graph.get_all_nodes()
    # Simple fuzzy matching - first 3 characters
    prefix = target[:3].lower() if len(target) >= 3 else target.lower()
    return [n['name'] for n in all_nodes if n['name'].lower().startswith(prefix)][:5]
```

### CLI Implementation

```python
# saw/cli/impact.py
"""CLI commands for impact analysis."""
import asyncio
import json
import click
from rich.console import Console
from rich.table import Table
from rich.panel import Panel

from saw.analysis.impact import analyze_impact, NodeNotFoundError
from saw.graph import get_graph


console = Console()


@click.group()
def impact_group():
    """Impact analysis commands."""
    pass


@impact_group.command()
@click.argument('target')
@click.option('--direction', '-d', type=click.Choice(['upstream', 'downstream']),
              default='upstream', help='Analysis direction')
@click.option('--max-depth', '-m', type=int, default=3,
              help='Maximum traversal depth')
@click.option('--min-confidence', '-c', type=float, default=0.8,
              help='Minimum confidence threshold')
@click.option('--json', '-j', 'json_output', is_flag=True,
              help='Output as JSON')
@click.option('--include-tests', is_flag=False,
              help='Include test files')
def impact(target, direction, max_depth, min_confidence, json_output, include_tests):
    """
    Analyze code modification impact.

    Examples:
        saw impact UserService
        saw impact handleLogin --direction downstream
        saw impact AuthModule --max-depth 5 --min-confidence 0.9
    """
    graph = get_graph()

    try:
        result = asyncio.run(analyze_impact(
            graph, target, direction, max_depth,
            min_confidence, None, include_tests
        ))

        if json_output:
            click.echo(json.dumps(result, indent=2))
        else:
            _print_impact_report(result)

    except NodeNotFoundError as e:
        console.print(f"[red]Error:[/red] {e}")
        console.print("\n[yellow]Similar nodes found:[/yellow]")
        # Would show suggestions here
        raise click.Exit(1)


def _print_impact_report(result: dict):
    """Print formatted impact report."""
    target = result['target']
    direction = result['direction']
    summary = result['summary']

    # Header
    direction_text = "depends on" if direction == "downstream" else "is depended on by"
    console.print(Panel(
        f"Modifying [bold]{target}[/bold] {direction_text}",
        title="Impact Analysis",
        style="blue"
    ))

    # Summary
    console.print(f"\n[cyan]Summary:[/cyan]")
    console.print(f"  Total affected: {summary['total_affected']}")
    console.print(f"  Depth 1 (will break): {summary['depth_1_count']}")
    console.print(f"  Depth 2 (likely affected): {summary['depth_2_count']}")
    console.print(f"  Depth 3 (may need testing): {summary['depth_3_count']}")

    if summary['high_risk_count'] > 0:
        console.print(f"\n[red bold]⚠ HIGH RISK: {summary['high_risk_count']} direct dependents![/red bold]")

    # Impact table
    if result['impacts']:
        table = Table(title="Affected Nodes")
        table.add_column("Depth", style="cyan", width=6)
        table.add_column("Risk", style="red", width=15)
        table.add_column("Name", style="white")
        table.add_column("Type", style="dim")
        table.add_column("File", style="dim")
        table.add_column("Confidence", width=10)

        for impact in result['impacts'][:20]:  # Limit display
            risk_style = {
                'WILL_BREAK': 'red bold',
                'LIKELY_AFFECTED': 'yellow',
                'MAY_NEED_TESTING': 'blue'
            }.get(impact['risk_level'], 'white')

            table.add_row(
                str(impact['depth']),
                f"[{risk_style}]{impact['risk_level']}[/{risk_style}]",
                impact['name'],
                impact['kind'],
                impact['file_path'].split('/')[-1],
                f"{impact['confidence']:.0%}"
            )

        console.print(table)

        if len(result['impacts']) > 20:
            console.print(f"\n[dim]... and {len(result['impacts']) - 20} more[/dim]")

    # Execution time
    console.print(f"\n[dim]Analysis completed in {result['execution_time_ms']:.2f}ms[/dim]")
```

# tests/integration/test_impact_cli.py
import pytest
from click.testing import CliRunner
from saw.cli.impact import impact


class TestImpactCLI:
    def test_basic_impact(self):
        """Test basic impact command."""
        runner = CliRunner()
        result = runner.invoke(impact, ['UserService'])

        assert result.exit_code == 0
        assert 'Impact Analysis' in result.output

    def test_direction_option(self):
        """Test direction option."""
        runner = CliRunner()
        result = runner.invoke(impact, ['UserService', '--direction', 'downstream'])

        assert result.exit_code == 0

    def test_json_output(self):
        """Test JSON output format."""
        runner = CliRunner()
        result = runner.invoke(impact, ['UserService', '--json'])

        assert result.exit_code == 0
        import json
        data = json.loads(result.output)
        assert 'target' in data
        assert 'impacts' in data

    def test_node_not_found(self):
        """Test error on missing node."""
        runner = CliRunner()
        result = runner.invoke(impact, ['NonExistentNode'])

        assert result.exit_code == 1
        assert 'Error' in result.output
