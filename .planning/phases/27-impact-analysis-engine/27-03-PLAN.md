```python
# saw/api/impact.py
"""API endpoints for impact analysis."""
from fastapi import APIRouter, HTTPException, Query
from pydantic import BaseModel

from saw.analysis.impact import analyze_impact, NodeNotFoundError


router = APIRouter(prefix="/api/impact", tags=["impact"])


class ImpactRequest(BaseModel):
    target: str
    direction: str = "upstream"
    max_depth: int = 3
    min_confidence: float = 0.8
    relation_types: list[str] = None
    include_tests: bool = False


@router.post("/")
async def analyze_impact_endpoint(request: ImpactRequest):
    """
    Analyze code modification impact.

    Returns nodes that will be affected by modifying the target.
    """
    graph = get_current_graph()

    try:
        result = analyze_impact(
            graph,
            request.target,
            request.direction,
            request.max_depth,
            request.min_confidence,
            request.relation_types,
            request.include_tests
        )
        return result

    except NodeNotFoundError:
        raise HTTPException(
            status_code=404,
            detail=f"Node '{request.target}' not found"
        )


@router.get("/{target}")
async def get_impact(
    target: str,
    direction: str = Query("upstream", regex="^(upstream|downstream)$"),
    max_depth: int = Query(3, ge=1, le=5),
    min_confidence: float = Query(0.8, ge=0.0, le=1.0)
):
    """
    Get impact analysis for a target symbol.

    - **target**: Symbol name or UID
    - **direction**: 'upstream' (dependents) or 'downstream' (dependencies)
    - **max_depth**: Traversal depth limit (1-5)
    - **min_confidence**: Minimum confidence (0.0-1.0)
    """
    graph = get_current_graph()

    try:
        return analyze_impact(
            graph, target, direction, max_depth, min_confidence
        )
    except NodeNotFoundError:
        raise HTTPException(
            status_code=404,
            detail=f"Node '{target}' not found"
        )


@router.get("/{target}/graph")
async def get_impact_graph(target: str, direction: str = "upstream"):
    """
    Get impact analysis as graph data for visualization.

    Returns nodes and edges in a format suitable for D3.js.
    """
    graph = get_current_graph()

    try:
        result = analyze_impact(graph, target, direction)

        # Convert to D3.js format
        nodes = []
        edges = []

        # Add target node
        nodes.append({
            "id": result["target_node"]["uid"],
            "name": result["target"],
            "type": "target",
            "risk": "TARGET"
        })

        # Add impact nodes
        for impact in result["impacts"]:
            nodes.append({
                "id": impact["uid"],
                "name": impact["name"],
                "type": impact["kind"],
                "risk": impact["risk_level"],
                "depth": impact["depth"],
                "confidence": impact["confidence"]
            })

            # Add edge
            if direction == "upstream":
                edges.append({
                    "source": impact["uid"],
                    "target": result["target_node"]["uid"],
                    "type": impact["relation_type"],
                    "confidence": impact["confidence"]
                })
            else:
                edges.append({
                    "source": result["target_node"]["uid"],
                    "target": impact["uid"],
                    "type": impact["relation_type"],
                    "confidence": impact["confidence"]
                })

        return {"nodes": nodes, "links": edges, "summary": result["summary"]}

    except NodeNotFoundError:
        raise HTTPException(
            status_code=404,
            detail=f"Node '{target}' not found"
        )
```

```vue
<!-- frontend/src/components/ImpactGraph.vue -->
<template>
  <div class="impact-graph">
    <!-- Controls -->
    <div class="controls">
      <div class="search">
        <input
          v-model="target"
          placeholder="Enter symbol name..."
          @keyup.enter="analyze"
        />
        <button @click="analyze">Analyze</button>
      </div>

      <div class="options">
        <select v-model="direction">
          <option value="upstream">Upstream (Dependents)</option>
          <option value="downstream">Downstream (Dependencies)</option>
        </select>

        <input
          type="range"
          v-model="maxDepth"
          min="1"
          max="5"
        />
        <span>Depth: {{ maxDepth }}</span>

        <input
          type="range"
          v-model="minConfidence"
          min="0"
          max="100"
        />
        <span>Confidence: {{ minConfidence }}%</span>
      </div>
    </div>

    <!-- Summary -->
    <div v-if="summary" class="summary">
      <div class="stat high-risk" v-if="summary.high_risk_count > 0">
        ⚠ {{ summary.high_risk_count }} WILL BREAK
      </div>
      <div class="stat">{{ summary.depth_1_count }} Direct</div>
      <div class="stat">{{ summary.depth_2_count }} Indirect</div>
      <div class="stat">{{ summary.depth_3_count }} Potential</div>
    </div>

    <!-- Graph -->
    <div ref="graphContainer" class="graph-container"></div>

    <!-- Legend -->
    <div class="legend">
      <div class="legend-item high-risk">
        <span class="dot"></span> Will Break
      </div>
      <div class="legend-item medium-risk">
        <span class="dot"></span> Likely Affected
      </div>
      <div class="legend-item low-risk">
        <span class="dot"></span> May Need Testing
      </div>
    </div>
  </div>
</template>

<script>
import * as d3 from 'd3'
import { ref, onMounted } from 'vue'

export default {
  setup() {
    const target = ref('')
    const direction = ref('upstream')
    const maxDepth = ref(3)
    const minConfidence = ref(80)
    const summary = ref(null)
    const graphContainer = ref(null)

    const analyze = async () => {
      if (!target.value) return

      const response = await fetch(
        `/api/impact/${target.value}/graph?direction=${direction.value}` +
        `&max_depth=${maxDepth.value}` +
        `&min_confidence=${minConfidence.value / 100}`
      )

      const data = await response.json()
      summary.value = data.summary
      renderGraph(data)
    }

    const renderGraph = (data) => {
      // D3.js force simulation
      const width = 800
      const height = 600

      const svg = d3.select(graphContainer.value)
        .append('svg')
        .attr('width', width)
        .attr('height', height)

      const simulation = d3.forceSimulation(data.nodes)
        .force('link', d3.forceLink(data.links).id(d => d.id))
        .force('charge', d3.forceManyBody().strength(-200))
        .force('center', d3.forceCenter(width / 2, height / 2))

      // Color by risk level
      const colorScale = d3.scaleOrdinal()
        .domain(['WILL_BREAK', 'LIKELY_AFFECTED', 'MAY_NEED_TESTING', 'TARGET'])
        .range(['#ff4444', '#ffaa00', '#4488ff', '#00aa00'])

      // Draw links
      const link = svg.append('g')
        .selectAll('line')
        .data(data.links)
        .enter()
        .append('line')
        .attr('stroke', '#666')
        .attr('stroke-width', d => d.confidence * 3)

      // Draw nodes
      const node = svg.append('g')
        .selectAll('circle')
        .data(data.nodes)
        .enter()
        .append('circle')
        .attr('r', 10)
        .attr('fill', d => colorScale(d.risk))
        .call(d3.drag()
          .on('start', dragStarted)
          .on('drag', dragged)
          .on('end', dragEnded))

      // Labels
      node.append('title').text(d => d.name)

      simulation.on('tick', () => {
        link
          .attr('x1', d => d.source.x)
          .attr('y1', d => d.source.y)
          .attr('x2', d => d.target.x)
          .attr('y2', d => d.target.y)

        node
          .attr('cx', d => d.x)
          .attr('cy', d => d.y)
      })
    }

    return {
      target, direction, maxDepth, minConfidence,
      summary, graphContainer, analyze
    }
  }
}
</script>
```
