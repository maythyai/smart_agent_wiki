```python
"""Impact analysis types."""
from typing import TypedDict, Literal, Optional
from datetime import datetime

RiskLevel = Literal['WILL_BREAK', 'LIKELY_AFFECTED', 'MAY_NEED_TESTING']
Direction = Literal['upstream', 'downstream']


class ImpactNode(TypedDict):
    """Single affected node."""
    uid: str
    name: str
    kind: str
    file_path: str
    start_line: int
    depth: int
    risk_level: RiskLevel
    relation_type: str
    confidence: float


class ImpactResult(TypedDict):
    """Impact analysis result."""
    target: str
    target_node: dict
    direction: Direction
    impacts: list[ImpactNode]
    summary: dict
    execution_time_ms: float
    analyzed_at: str
```

### Step 2: Core Algorithm (impact.py)

```python
"""Impact analysis core algorithm."""
from __future__ import annotations
import time
from collections import deque
from typing import Optional

from .types import ImpactNode, ImpactResult, RiskLevel, Direction


def analyze_impact(
    graph,
    target: str,
    direction: Direction = 'upstream',
    max_depth: int = 3,
    min_confidence: float = 0.5,
    relation_types: list[str] = None,
    include_tests: bool = False
) -> ImpactResult:
    """
    GitNexus-style impact analysis.

    Algorithm:
    1. Find target node by name or UID
    2. BFS traverse along specified edges
    3. Group by depth with risk labels
    4. Filter by confidence threshold
    5. Return structured result
    """
    relation_types = relation_types or ['CALLS', 'IMPORTS', 'EXTENDS', 'IMPLEMENTS']

    start = time.time()

    # Find target
    target_node = _find_node(graph, target)
    if not target_node:
        raise NodeNotFoundError(f"Node '{target}' not found")

    # BFS traversal
    visited = set()
    impacts = []
    queue = deque([(target_node['uid'], 0)])

    while queue:
        node_id, depth = queue.popleft()

        if depth > max_depth:
            continue

        if node_id in visited:
            continue

        visited.add(node_id)

        # Get edges based on direction
        if direction == 'upstream':
            edges = graph.get_incoming_edges(node_id, relation_types)
        else:
            edges = graph.get_outgoing_edges(node_id, relation_types)

        for edge in edges:
            if edge['confidence'] < min_confidence:
                continue

            # Get the dependent node
            dep_id = edge['source'] if direction == 'upstream' else edge['target']
            dep_node = graph.get_node(dep_id)

            if dep_node is None:
                continue

            # Filter tests if needed
            if not include_tests and _is_test_node(dep_node):
                continue

            # Add to results
            impacts.append(ImpactNode(
                uid=dep_id,
                name=dep_node['name'],
                kind=dep_node['kind'],
                file_path=dep_node['filePath'],
                start_line=dep_node['startLine'],
                depth=depth + 1,
                risk_level=_get_risk_level(depth + 1),
                relation_type=edge['type'],
                confidence=edge['confidence']
            ))

            queue.append((dep_id, depth + 1))

    # Sort by depth, then confidence
    impacts.sort(key=lambda x: (x['depth'], -x['confidence']))

    execution_time_ms = (time.time() - start) * 1000

    return ImpactResult(
        target=target,
        target_node=target_node,
        direction=direction,
        impacts=impacts,
        summary=_get_summary(impacts),
        execution_time_ms=execution_time_ms,
        analyzed_at=datetime.utcnow().isoformat()
    )


def _find_node(graph, target: str) -> Optional[dict]:
    """Find node by name or UID."""
    # Try exact UID match first
    node = graph.get_node(target)
    if node:
        return node

    # Try name lookup
    nodes = graph.find_nodes_by_name(target)
    if nodes:
        return nodes[0]  # Return first match

    return None


def _get_risk_level(depth: int) -> RiskLevel:
    """Map depth to risk level."""
    if depth == 1:
        return 'WILL_BREAK'
    elif depth == 2:
        return 'LIKELY_AFFECTED'
    else:
        return 'MAY_NEED_TESTING'


def _get_summary(impacts: list[ImpactNode]) -> dict:
    """Generate summary statistics."""
    return {
        'depth_1_count': sum(1 for i in impacts if i['depth'] == 1),
        'depth_2_count': sum(1 for i in impacts if i['depth'] == 2),
        'depth_3_count': sum(1 for i in impacts if i['depth'] == 3),
        'high_risk_count': sum(1 for i in impacts if i['risk_level'] == 'WILL_BREAK'),
        'total_affected': len(impacts)
    }


def _is_test_node(node: dict) -> bool:
    """Check if node is a test."""
    name = node.get('name', '').lower()
    file_path = node.get('filePath', '').lower()

    test_indicators = ['test', 'spec', '_test', '_spec', 'tests/', 'test_']
    return any(ind in name or ind in file_path for ind in test_indicators)


class NodeNotFoundError(Exception):
    """Node not found in knowledge graph."""
    pass
```

```python
# tests/unit/analysis/test_impact.py
import pytest
from saw.analysis.impact import analyze_impact, NodeNotFoundError


class TestImpactAnalysis:
    def test_upstream_simple(self, mock_graph):
        """Test upstream analysis finds dependents."""
        result = analyze_impact(mock_graph, 'UserService', direction='upstream')

        assert result['target'] == 'UserService'
        assert result['direction'] == 'upstream'
        assert len(result['impacts']) > 0

        # Check depth 1 are WILL_BREAK
        depth_1 = [i for i in result['impacts'] if i['depth'] == 1]
        for i in depth_1:
            assert i['risk_level'] == 'WILL_BREAK'

    def test_downstream_analysis(self, mock_graph):
        """Test downstream analysis finds dependencies."""
        result = analyze_impact(mock_graph, 'UserService', direction='downstream')

        assert result['direction'] == 'downstream'
        # Should find what UserService depends on

    def test_max_depth_limit(self, mock_graph):
        """Test max_depth limits traversal."""
        result = analyze_impact(mock_graph, 'UserService', max_depth=1)

        max_depth = max(i['depth'] for i in result['impacts'])
        assert max_depth <= 1

    def test_confidence_filter(self, mock_graph):
        """Test min_confidence filters low confidence edges."""
        result = analyze_impact(mock_graph, 'UserService', min_confidence=0.8)

        for impact in result['impacts']:
            assert impact['confidence'] >= 0.8

    def test_node_not_found(self, mock_graph):
        """Test NodeNotFoundError on missing node."""
        with pytest.raises(NodeNotFoundError):
            analyze_impact(mock_graph, 'NonExistentNode')

    def test_summary_statistics(self, mock_graph):
        """Test summary includes correct counts."""
        result = analyze_impact(mock_graph, 'UserService')

        summary = result['summary']
        total = summary['depth_1_count'] + summary['depth_2_count'] + summary['depth_3_count']
        assert total == summary['total_affected']
```
