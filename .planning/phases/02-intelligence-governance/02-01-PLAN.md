From src/saw/domain/value_objects.py:

```python
class ConfidenceLevel(IntEnum):
    UNVERIFIED = 1      # LLM generated, no validation
    SINGLE_SOURCE = 2   # Single source extracted
    CROSS_VALIDATED = 3 # Multiple sources agree
    HUMAN_VERIFIED = 4  # Human reviewed and approved

class SourceMark(IntEnum):
    EXTRACTED = 1   # Directly extracted from source
    INFERRED = 2    # Inferred from context
    AMBIGUOUS = 3   # Uncertain origin

class FreshnessLevel(IntEnum):
    LEVEL_0 = 0  # Just created
    LEVEL_1 = 1  # 1 day old
    LEVEL_2 = 2  # 3 days old
    LEVEL_3 = 3  # 1 week old (yellow)
    LEVEL_4 = 4  # 2 weeks old (yellow)
    LEVEL_5 = 5  # 1 month old (orange)
    LEVEL_6 = 6  # 3 months old (orange)
    LEVEL_7 = 7  # 6 months old (red)
    LEVEL_8 = 8  # Over 6 months (red)
```
