    ```typescript
    // Extension settings (reused pattern from Obsidian plugin)
    export interface ExtensionSettings {
      apiUrl: string;
      apiToken: string;
      autoSync: boolean;
      showNotifications: boolean;
      lastClipTime?: string;
    }

    export const DEFAULT_SETTINGS: ExtensionSettings = {
      apiUrl: 'http://localhost:8000',
      apiToken: '',
      autoSync: true,
      showNotifications: true,
    };

    // Message types for communication
    export type MessageType =
      | 'extract-page'
      | 'extract-selection'
      | 'parse-readability'
      | 'clip-page'
      | 'clip-selection'
      | 'get-settings'
      | 'update-settings'
      | 'get-history'
      | 'content-ready';

    export interface Message<T = unknown> {
      type: MessageType;
      data?: T;
      tabId?: number;
    }

      clippedAt: string;

    // API response types (matching backend)
    export interface ClipResponse {
      status: 'success' | 'error';
      id?: string;
      error?: string;
    }

    export interface TagSuggestion {
      tags: string[];
      confidence: number;
    }
    ```

<task type="auto">
  <name>Task 4: Create storage manager for Chrome storage API</name>
  <files>
    extensions/chrome-clipper/src/background/storage.ts
  </files>
  <action>
    Implement StorageManager following Pitfall 22 (all state persisted to chrome.storage) and Pitfall 24 (use storage.local for large data).

    **StorageManager class:**
    - `getSettings(): Promise<ExtensionSettings>` — Load from storage.local with defaults
    - `updateSettings(partial: Partial<ExtensionSettings>): Promise<void>` — Merge and save
    - `getClipHistory(): Promise<ClippedContent[]>` — Last 100 clips stored in storage.local
    - `addToHistory(clip: ClippedContent): Promise<void>` — Append with size limit check
    - `clearHistory(): Promise<void>` — Clear clip history
    - `getPendingSync(): Promise<ClippedContent[]>` — Offline queue for failed uploads
    - `addToPendingSync(clip: ClippedContent): Promise<void>` — Queue for retry
    - `removeFromPendingSync(id: string): Promise<void>` — Remove after successful sync

    **Key implementation notes:**
    - Use chrome.storage.local (not sync) — per Pitfall 24
    - Handle chrome.runtime.lastError on all storage operations
    - Serialize Date objects as ISO strings
    - Implement size checking before writes (10MB limit)

    **Avoid:** Don't use chrome.storage.sync — quota is 100KB, clips exceed this easily.
  </action>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User Browser → Extension | Untrusted web content passes through content script |
| Extension → SAW API | Authenticated requests to backend server |
| Chrome Storage → Extension | Persisted settings and clip history |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-01 | Tampering | storage.ts | mitigate | Validate all data read from chrome.storage before use |
| T-08-02 | Information Disclosure | manifest.json | mitigate | host_permissions limits to known SAW instances only |
| T-08-03 | Denial of Service | background/index.ts | accept | Service worker lifecycle managed by Chrome |
| T-08-04 | Elevation of Privilege | messaging.ts | mitigate | Validate message sender before processing sensitive operations |
