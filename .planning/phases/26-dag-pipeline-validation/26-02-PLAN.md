```python
    """Legacy pipeline implementation."""

    # Stage 2: Parse
    content = await parse(source, file_type)
    
    # Stage 3: Extract
    claims = await extract(content)
    
    # Stage 4: Merge
    merged = await merge(claims)
    
    # Stage 5: Validate
    validated = await validate(merged)
    
    # Stage 6: Store
```

### Step 2: Implement Phase 1 (Classify)

```python
# saw/ingest/pipeline/phases/classify.py

from saw.ingest.pipeline.types import PipelinePhase, PipelineContext, PhaseResults
from saw.ingest.pipeline.phases import ClassifyOutput
from saw.ingest.classify import detect_file_type, extract_metadata


async def classify_execute(
    ctx: PipelineContext, 
    deps: PhaseResults
) -> ClassifyOutput:
    """
    Classify the source file and extract basic metadata.
    
    Phase 1: No dependencies
    """
    source = ctx.repo_path
    
    # Detect file type
    file_type = await detect_file_type(source)
    
    # Extract basic metadata
    metadata = await extract_metadata(source, file_type)
    
    # Detect language if code file
    detected_language = None
    if file_type in ('python', 'javascript', 'typescript', 'rust', 'go'):
        detected_language = file_type
    
    ctx.report_progress('classify', 1.0)
    
    return ClassifyOutput(
        file_type=file_type,
        source_path=source,
        metadata=metadata,
        detected_language=detected_language
    )


# Phase definition
CLASSIFY_PHASE = PipelinePhase(
    name='classify',
    deps=[],
    execute=classify_execute,
    description='Classify source file type and extract metadata'
)
```

```python
# saw/ingest/pipeline/phases/parse.py

from saw.ingest.pipeline.types import PipelinePhase, PipelineContext, PhaseResults
from saw.ingest.pipeline.phases import ParseOutput, ClassifyOutput
from saw.ingest.parse import parse_content
import time


async def parse_execute(
    ctx: PipelineContext,
    deps: PhaseResults
) -> ParseOutput:
    """
    Parse the source content.
    
    Phase 2: Depends on classify
    """
    import time
    
    # Get classify output (type-safe)
    classify_output: ClassifyOutput = deps.get_output('classify')
    
    source = classify_output['source_path']
    file_type = classify_output['file_type']
    
    start = time.time()
    
    # Parse content based on file type
    result = await parse_content(source, file_type)
    
    parse_time_ms = (time.time() - start) * 1000
    
    ctx.report_progress('parse', 1.0)
    
    return ParseOutput(
        raw_content=result['content'],
        sections=result.get('sections', []),
        metadata=classify_output['metadata'],
        parse_time_ms=parse_time_ms
    )


# Phase definition
PARSE_PHASE = PipelinePhase(
    name='parse',
    deps=['classify'],
    execute=parse_execute,
    description='Parse source content into structured format'
)
```

```python
from saw.ingest.pipeline.types import PipelinePhase, PipelineContext, PhaseResults
from saw.ingest.pipeline.phases import ExtractOutput, ParseOutput
from saw.ingest.extract import extract_claims, extract_entities, extract_relations
import time


async def extract_execute(
    ctx: PipelineContext,
    deps: PhaseResults
) -> ExtractOutput:
    """
    Extract claims, entities, and relations.
    
    Phase 3: Depends on parse
    """
    import time
    
    # Get parse output (type-safe)
    parse_output: ParseOutput = deps.get_output('parse')
    
    content = parse_output['raw_content']
    sections = parse_output['sections']
    
    start = time.time()
    
    # Extract claims
    claims = await extract_claims(content, sections)
    
    # Extract entities
    entities = await extract_entities(content)
    
    # Extract relations
    relations = await extract_relations(claims, entities)
    
    extraction_time_ms = (time.time() - start) * 1000
    
    ctx.report_progress('extract', 1.0)
    
    return ExtractOutput(
        claims=claims,
        entities=entities,
        relations=relations,
        extraction_time_ms=extraction_time_ms
    )


# Phase definition
EXTRACT_PHASE = PipelinePhase(
    name='extract',
    deps=['parse'],
    execute=extract_execute,
    description='Extract claims, entities, and relations from parsed content'
)
```

```python
from saw.ingest.pipeline.types import PipelinePhase, PipelineContext, PhaseResults

        merged_claims=merged_claims,
        conflicts=conflicts,
        merged_entities=entities
```

```python
from saw.ingest.pipeline.types import PipelinePhase, PipelineContext, PhaseResults

async def validate_execute(ctx: PipelineContext, deps: PhaseResults) -> ValidateOutput:

        validated_claims=validated_claims,
        validation_errors=errors,
        confidence_scores=confidence_scores
```

```python
# saw/ingest/pipeline/phases/store.py

from saw.ingest.pipeline.types import PipelinePhase, PipelineContext, PhaseResults
from saw.ingest.pipeline.phases import StoreOutput, ValidateOutput
from saw.ingest.store import store_to_vault, store_to_claims_db, create_wiki_pages


async def store_execute(ctx: PipelineContext, deps: PhaseResults) -> StoreOutput:
    """Store validated claims to vault and claims DB."""
    validate_output: ValidateOutput = deps.get_output('validate')
    
    claims = validate_output['validated_claims']
    
    # Store to vault (immutable)
    vault_path = await store_to_vault(claims, ctx.repo_path)
    
    # Store to claims DB
    claims_db_path = await store_to_claims_db(claims, ctx.graph)
    
    # Create wiki pages
    wiki_pages = await create_wiki_pages(claims, ctx.graph)
    
    ctx.report_progress('store', 1.0)
    
    return StoreOutput(
        vault_path=vault_path,
        claims_db_path=claims_db_path,
        stored_count=len(claims),
        wiki_pages_created=wiki_pages
    )


STORE_PHASE = PipelinePhase(
    name='store',
    deps=['validate'],
    execute=store_execute,
    description='Store claims to vault, claims DB, and wiki'
)
```

### Step 7: New Pipeline Entry Point

```python
# saw/ingest/pipeline_v2.py

"""New DAG-based pipeline implementation."""
from saw.ingest.pipeline.runner import PipelineRunner
from saw.ingest.pipeline.phases import get_default_phase_list
from saw.ingest.graph import KnowledgeGraph


async def run_ingest_pipeline(
    source_path: str,
    options: dict = None,
    progress_callback = None
) -> dict:
    """
    Run the DAG-based ingest pipeline.
    
    Args:
        source_path: Path to source file/directory
        options: Optional pipeline options
        progress_callback: Optional progress callback
        
    Returns:
        Pipeline results with all phase outputs
    """
    options = options or {}
    
    # Create context
    graph = KnowledgeGraph()
    ctx = PipelineContext(
        graph=graph,
        repo_path=source_path,
        options=options,
        progress_callback=progress_callback
    )
    
    # Get phases
    phases = get_default_phase_list()
    
    # Create runner
    runner = PipelineRunner(phases)
    
    # Execute
    results = await runner.run(ctx)
    
    # Return results
    return {
        'graph': graph,
        'phases': {name: results.get(name).output for name in results.names()},
        'execution_order': runner.get_execution_order(),
    }
```
