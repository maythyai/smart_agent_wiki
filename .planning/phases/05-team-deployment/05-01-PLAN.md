   ```python
   from .models import User, Vault, Claim, AuditLog
   ```

2. 创建 `src/saw/db/base.py`：
   ```python
   from sqlalchemy import create_engine
   from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
   from sqlalchemy.orm import sessionmaker, declarative_base

   Base = declarative_base()

   def get_engine(database_url: str, async_mode: bool = False):
       if async_mode:
           return create_async_engine(database_url, echo=False)
       return create_engine(database_url, echo=False)
   ```

3. 创建 `src/saw/db/session.py`：
   ```python
   from contextlib import asynccontextmanager
   from sqlalchemy.ext.asyncio import AsyncSession

   @asynccontextmanager
   async def get_session():
       async with AsyncSession(engine) as session:
           yield session
   ```

4. 更新 `src/saw/config.py` 添加数据库配置：
   ```python
   @dataclass
   class DatabaseConfig:
       url: str = "sqlite:///saw.db"
       pool_size: int = 5
       max_overflow: int = 10
       echo: bool = False
   ```
</action>


2. 更新 `src/saw/db/base.py` 支持 PostgreSQL URL：
   ```python
   def get_database_url(config: DatabaseConfig) -> str:
       if config.url.startswith("postgresql"):
           # 异步模式使用 postgresql+asyncpg
           return config.url.replace("postgresql://", "postgresql+asyncpg://")
       return config.url
   ```

3. 添加连接池配置：
   ```python
   engine = create_async_engine(
       database_url,
       pool_size=config.database.pool_size,
       max_overflow=config.database.max_overflow,
       pool_pre_ping=True,  # 检查连接有效性
   )
   ```
