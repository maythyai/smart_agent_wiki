3. 创建 `src/saw/cache/redis_cache.py`：
   ```python
   import redis.asyncio as redis
   import json
   from typing import Optional, Any
   from datetime import timedelta

   class RedisCache:
       def __init__(self, url: str):
           self.client = redis.from_url(url)
           self.prefix = "saw:"

       async def get(self, key: str) -> Optional[Any]:
           value = await self.client.get(f"{self.prefix}{key}")
           if value:
               return json.loads(value)
           return None

       async def set(
           self,
           key: str,
           value: Any,
           ttl: Optional[timedelta] = None
       ) -> None:
           serialized = json.dumps(value)
           if ttl:
               await self.client.setex(
                   f"{self.prefix}{key}",
                   int(ttl.total_seconds()),
                   serialized
               )
           else:
               await self.client.set(f"{self.prefix}{key}", serialized)

       async def delete(self, key: str) -> None:
           await self.client.delete(f"{self.prefix}{key}")

       async def invalidate_pattern(self, pattern: str) -> None:
           keys = await self.client.keys(f"{self.prefix}{pattern}*")
           if keys:
               await self.client.delete(*keys)

       async def close(self) -> None:
           await self.client.close()

   # 缓存装饰器
   def cache_result(ttl: timedelta = timedelta(minutes=5)):
       def decorator(func):
           async def wrapper(self, *args, **kwargs):
               cache_key = f"{func.__name__}:{str(args)}:{str(kwargs)}"
               cached = await self.cache.get(cache_key)
               if cached:
                   return cached
               result = await func(self, *args, **kwargs)
               await self.cache.set(cache_key, result, ttl)
               return result
           return wrapper
       return decorator
   ```

1. 创建 `src/saw/api/routes/health.py`：
   ```python
   from fastapi import APIRouter, Response
   from datetime import datetime
   import sqlalchemy as sa
   from saw import __version__

   router = APIRouter(tags=["health"])

   @router.get("/health")
   async def health_check():
       """基础健康检查"""
       return {
           "status": "healthy",
           "version": __version__,
           "timestamp": datetime.utcnow().isoformat()
       }

   @router.get("/health/live")
   async def liveness():
       """Kubernetes 存活探针"""
       return {"status": "alive"}

   @router.get("/health/ready")
   async def readiness(response: Response):
       """Kubernetes 就绪探针 - 检查所有依赖"""
       checks = {}

       # 检查数据库
       try:
           async with get_session() as session:
               await session.execute(sa.text("SELECT 1"))
           checks["database"] = "healthy"
       except Exception as e:
           checks["database"] = f"unhealthy: {str(e)}"
           response.status_code = 503

       # 检查 Redis
       try:
           cache = get_redis_cache()
           await cache.client.ping()
           checks["redis"] = "healthy"
       except Exception as e:
           checks["redis"] = f"unhealthy: {str(e)}"
           response.status_code = 503

       return {
           "status": "ready" if response.status_code != 503 else "not_ready",
           "checks": checks
       }

   @router.get("/metrics")
   async def metrics():
       """Prometheus 指标端点"""
       # 返回基础指标
       return {
           "saw_info": {
               "version": __version__,
           },
           "saw_users_total": await get_user_count(),
           "saw_vaults_total": await get_vault_count(),
           "saw_claims_total": await get_claim_count(),
       }
   ```

### Task 5: Implement Backup & Restore

<read_first>
- docker/docker-compose.yml (Docker 配置)
</read_first>

<action>
实现备份和恢复功能：

1. 创建 `scripts/backup.sh`：
   ```bash
   #!/bin/bash
   set -e

   BACKUP_DIR="/app/data/backups"
   TIMESTAMP=$(date +%Y%m%d_%H%M%S)
   BACKUP_FILE="${BACKUP_DIR}/saw_backup_${TIMESTAMP}.tar.gz"

   mkdir -p $BACKUP_DIR

   # 备份 PostgreSQL 数据库
   docker exec saw-db pg_dump -U saw saw > "${BACKUP_DIR}/db_${TIMESTAMP}.sql"

   # 备份 Redis 数据
   docker exec saw-redis redis-cli SAVE
   docker cp saw-redis:/data/dump.rdb "${BACKUP_DIR}/redis_${TIMESTAMP}.rdb"

   # 备份 Vault 文件
   tar -czf "${BACKUP_DIR}/vaults_${TIMESTAMP}.tar.gz" -C /app/data vaults

   # 创建完整备份包
   tar -czf $BACKUP_FILE \
       -C $BACKUP_DIR \
       "db_${TIMESTAMP}.sql" \
       "redis_${TIMESTAMP}.rdb" \
       "vaults_${TIMESTAMP}.tar.gz"

   # 清理临时文件
   rm "${BACKUP_DIR}/db_${TIMESTAMP}.sql"
   rm "${BACKUP_DIR}/redis_${TIMESTAMP}.rdb"
   rm "${BACKUP_DIR}/vaults_${TIMESTAMP}.tar.gz"

   # 保留最近 7 天的备份
   find $BACKUP_DIR -name "saw_backup_*.tar.gz" -mtime +7 -delete

   echo "Backup created: $BACKUP_FILE"
   ```

2. 创建 `scripts/restore.sh`：
   ```bash
   #!/bin/bash
   set -e

   if [ -z "$1" ]; then
       echo "Usage: ./restore.sh <backup_file>"
       exit 1
   fi

   BACKUP_FILE=$1
   TEMP_DIR=$(mktemp -d)

   # 解压备份
   tar -xzf $BACKUP_FILE -C $TEMP_DIR

   # 恢复数据库
   docker exec -i saw-db psql -U saw saw < "${TEMP_DIR}/db_*.sql"

   # 恢复 Redis
   docker cp "${TEMP_DIR}/redis_*.rdb" saw-redis:/data/dump.rdb
   docker restart saw-redis

   # 恢复 Vaults
   tar -xzf "${TEMP_DIR}/vaults_*.tar.gz" -C /app/data

   # 清理
   rm -rf $TEMP_DIR

   echo "Restore completed from: $BACKUP_FILE"
   ```

3. 创建 `scripts/backup_cron`：
   ```
   # 每天凌晨 2 点执行备份
   0 2 * * * /app/scripts/backup.sh >> /var/log/saw_backup.log 2>&1
   ```
</action>
