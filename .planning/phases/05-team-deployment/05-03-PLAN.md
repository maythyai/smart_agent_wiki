   ```python
       READ = "read"
       WRITE = "write"
       DELETE = "delete"
       ADMIN = "admin"
   ```

2. 实现 `VaultPermissionManager` 类：
   ```python
   class VaultPermissionManager:
       def __init__(self, session: AsyncSession, cedar: CedarClient):
           self.session = session
           self.cedar = cedar

       async def grant_permission(
           self,
           vault_id: str,
           user_id: str,
           permission: Permission
       ) -> None:
           # 添加到 vault_permissions 表
           vp = VaultPermission(
               vault_id=vault_id,
               user_id=user_id,
               permission=permission.value
           )
           self.session.add(vp)
           await self.session.commit()

       async def revoke_permission(
           self,
           vault_id: str,
           user_id: str
       ) -> None:
           # 从 vault_permissions 表删除
           await self.session.execute(
               delete(VaultPermission)
               .where(VaultPermission.vault_id == vault_id)
               .where(VaultPermission.user_id == user_id)
           )
           await self.session.commit()

       async def check_permission(
           self,
           vault_id: str,
           user_id: str,
           action: str
       ) -> bool:
           # 1. 检查是否是 owner
           vault = await self.session.get(Vault, vault_id)
           if vault and vault.owner_id == user_id:
               return True

           # 2. 检查 Cedar 策略
           return await check_vault_permission(
               user_id, vault_id, action, self.cedar
           )

       async def list_permissions(
           self,
           vault_id: str
       ) -> List[VaultPermission]:
           result = await self.session.execute(
               select(VaultPermission)
               .where(VaultPermission.vault_id == vault_id)
           )
           return result.scalars().all()
   ```

3. 实现私有 Vault 检查：

   ```python
   async def is_private_vault(self, vault_id: str) -> bool:
       vault = await self.session.get(Vault, vault_id)
       return vault and not vault.is_shared
   ```
