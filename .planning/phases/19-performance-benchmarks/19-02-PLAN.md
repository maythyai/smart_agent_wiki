async def test_backpressure_queue_throttling(self, benchmark_dir):
    """PERF-07: Verify backpressure throttles at queue thresholds."""
    from saw.connectors.backpressure import (
        BackpressureManager, BackpressureConfig, BackpressureState
    )

    config = BackpressureConfig(
        pause_threshold=100,
        resume_threshold=50
    )

    # Create mock write queue
    mock_queue = MockWriteQueue()
    manager = BackpressureManager(mock_queue, config)

    # Fill queue to trigger pause
    mock_queue.fill_to(150)
    state = await manager.check()
    assert state == BackpressureState.PAUSED

    # Drain queue to trigger resume
    mock_queue.drain_to(30)
    state = await manager.check()
    assert state == BackpressureState.ACTIVE

    report = {
        "benchmark": "backpressure_throttling",
        "results": {
            "pause_triggered_at": 100,
            "resume_triggered_at": 50,
            "hysteresis_works": True
        }
    }
