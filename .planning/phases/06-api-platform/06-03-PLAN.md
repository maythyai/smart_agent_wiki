   ```python
   from fastapi import APIRouter, UploadFile, File, BackgroundTasks
   from typing import List

   router = APIRouter(prefix="/bulk", tags=["bulk"])

   @router.post("/import/vaults")
   async def bulk_import_vaults(
       file: UploadFile = File(...),
       background_tasks: BackgroundTasks = BackgroundTasks()
   ):
       """
       批量导入 Vaults（JSON 格式）
       文件格式：
       [
           {"name": "Vault 1", "entries": [...]},
           {"name": "Vault 2", "entries": [...]}
       ]
       """
       content = await file.read()
       data = json.loads(content)

       task_id = str(uuid4())

       # 后台处理
       background_tasks.add_task(
           process_bulk_import,
           task_id=task_id,
           data=data,
           user_id=get_current_user().id
       )

       return {
           "task_id": task_id,
           "status": "processing",
           "total_items": len(data)
       }

   @router.post("/import/claims/{vault_id}")
   async def bulk_import_claims(
       vault_id: str,
       file: UploadFile = File(...),
       background_tasks: BackgroundTasks
   ):
       """批量导入 Claims（JSON 或 CSV 格式）"""
       pass

   @router.get("/import/{task_id}/status")
   async def get_import_status(task_id: str):
       """获取导入任务状态"""
       status = await redis.get(f"bulk_import:{task_id}")
       return status or {"status": "not_found"}

   async def process_bulk_import(task_id: str, data: list, user_id: str):
       """后台处理批量导入"""
       total = len(data)
       processed = 0
       errors = []

       for item in data:
           try:
               await engine.import_item(item, user_id)
               processed += 1
           except Exception as e:
               errors.append({"item": item, "error": str(e)})

           # 更新进度
           await redis.set(
               f"bulk_import:{task_id}",
               json.dumps({
                   "status": "processing",
                   "processed": processed,
                   "total": total,
                   "errors": errors
               }),
               ex=3600
           )

       # 标记完成
       await redis.set(
           f"bulk_import:{task_id}",
           json.dumps({
               "status": "completed",
               "processed": processed,
               "total": total,
               "errors": errors
           }),
           ex=86400
       )
   ```
