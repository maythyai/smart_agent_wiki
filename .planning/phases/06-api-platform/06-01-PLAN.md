1. 创建 `src/saw/api_keys/dependencies.py`：
   ```python
   from fastapi import Depends, HTTPException, Header
   from sqlalchemy import select

   async def get_api_key(
       authorization: str = Header(None),
       session: AsyncSession = Depends(get_session)
   ) -> APIKey:
       if not authorization:
           raise HTTPException(status_code=401, detail="Missing Authorization header")

       # 支持 ApiKey 和 Bearer 两种格式
       if authorization.startswith("ApiKey "):
           key = authorization[7:]
       elif authorization.startswith("Bearer "):
           key = authorization[7:]  # JWT 在其他地方处理
           raise HTTPException(status_code=401, detail="Use ApiKey prefix for API keys")
       else:
           raise HTTPException(status_code=401, detail="Invalid Authorization format")

       # 查找 API Key
       key_hash = hash_api_key(key)
       result = await session.execute(
           select(APIKey).where(APIKey.key_hash == key_hash, APIKey.is_active == True)
       )
       api_key = result.scalar()

       if not api_key:
           raise HTTPException(status_code=401, detail="Invalid API key")

       # 检查过期
       if api_key.expires_at and api_key.expires_at < datetime.utcnow():
           raise HTTPException(status_code=401, detail="API key expired")

       # 更新最后使用时间
       api_key.last_used_at = datetime.utcnow()
       await session.commit()

       return api_key

   def hash_api_key(key: str) -> str:
       import hashlib
       return hashlib.sha256(key.encode()).hexdigest()
   ```

2. 创建权限检查依赖：

   def require_api_permission(*permissions: str):
       async def checker(
           api_key: APIKey = Depends(get_api_key)
       ) -> APIKey:
           key_perms = api_key.permissions.split(",")
           for perm in permissions:
               if perm not in key_perms:
                   raise HTTPException(status_code=403, detail=f"Missing permission: {perm}")
           return api_key
       return checker

1. 创建 `src/saw/api/middleware/rate_limit.py`：
   ```python
   from fastapi import Request, HTTPException
   from datetime import datetime, timedelta
   import time

   class RateLimiter:
       def __init__(self, redis: RedisCache):
           self.redis = redis

       async def check_rate_limit(
           self,
           key_id: str,
           limit_hour: int,
           limit_day: int
       ) -> tuple[bool, dict]:
           now = time.time()
           hour_key = f"ratelimit:{key_id}:hour:{int(now // 3600)}"
           day_key = f"ratelimit:{key_id}:day:{int(now // 86400)}"

           # 获取当前计数
           hour_count = await self.redis.client.incr(hour_key)
           day_count = await self.redis.client.incr(day_key)

           # 设置过期时间
           if hour_count == 1:
               await self.redis.client.expire(hour_key, 3600)
           if day_count == 1:
               await self.redis.client.expire(day_key, 86400)

           remaining_hour = max(0, limit_hour - hour_count)
           remaining_day = max(0, limit_day - day_count)

           return (
               hour_count <= limit_hour and day_count <= limit_day,
               {
                   "x-ratelimit-limit-hour": limit_hour,
                   "x-ratelimit-remaining-hour": remaining_hour,
                   "x-ratelimit-limit-day": limit_day,
                   "x-ratelimit-remaining-day": remaining_day,
                   "x-ratelimit-reset-hour": int(now + 3600),
                   "x-ratelimit-reset-day": int(now + 86400),
               }
           )

   @app.middleware("http")
   async def rate_limit_middleware(request: Request, call_next):
       # 检查是否有 API Key
       if "authorization" in request.headers:
           auth = request.headers["authorization"]
           if auth.startswith("ApiKey "):
               # 获取 API Key 信息
               api_key = await get_api_key_from_request(request)
               limiter = RateLimiter(get_redis())

               allowed, headers = await limiter.check_rate_limit(
                   api_key.id,
                   api_key.rate_limit_hour,
                   api_key.rate_limit_day
               )

               if not allowed:
                   raise HTTPException(status_code=429, detail="Rate limit exceeded", headers=headers)

               # 添加速率限制头到响应
               response = await call_next(request)
               for k, v in headers.items():
                   response.headers[k] = str(v)
               return response

       return await call_next(request)
   ```

   ```python
   from fastapi import APIRouter, Depends, HTTPException

   router = APIRouter(prefix="/api-keys", tags=["api-keys"])

   @router.post("/", response_model=APIKeyResponse)
   async def create_api_key(
       key_data: APIKeyCreate,
       user: User = Depends(get_current_active_user)
   ):
       # 生成 API Key
       raw_key = generate_api_key()
       key_hash = hash_api_key(raw_key)

       api_key = APIKey(
           id=str(uuid4()),
           key_hash=key_hash,
           name=key_data.name,
           user_id=user.id,
           permissions=",".join(key_data.permissions),
           rate_limit_hour=config.rate_limit.default_hour,
           rate_limit_day=config.rate_limit.default_day,
       )

       if key_data.expires_days:
           api_key.expires_at = datetime.utcnow() + timedelta(days=key_data.expires_days)

       session.add(api_key)
       await session.commit()

       # 仅在此处返回原始 key
       return APIKeyResponse(
           id=api_key.id,
           name=api_key.name,
           key=raw_key,
           permissions=key_data.permissions,
           rate_limit_hour=api_key.rate_limit_hour,
           created_at=api_key.created_at,
           expires_at=api_key.expires_at,
       )

   @router.get("/", response_model=List[APIKeyListResponse])
   async def list_api_keys(
       user: User = Depends(get_current_active_user)
   ):
       result = await session.execute(
           select(APIKey).where(APIKey.user_id == user.id)
       )
       return result.scalars().all()

   @router.delete("/{key_id}")
   async def revoke_api_key(
       key_id: str,
       user: User = Depends(get_current_active_user)
   ):
       result = await session.execute(
           select(APIKey).where(APIKey.id == key_id, APIKey.user_id == user.id)
       )
       api_key = result.scalar()
       if not api_key:
           raise HTTPException(status_code=404)

       api_key.is_active = False
       await session.commit()
       return {"status": "revoked"}
   ```
