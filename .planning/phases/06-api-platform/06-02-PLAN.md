   ```python
   import hmac
   import hashlib
   import json

   def sign_webhook(payload: dict, secret: str) -> str:
       body = json.dumps(payload, sort_keys=True, separators=(',', ':'))
       return hmac.new(
           secret.encode(),
           body.encode(),
           hashlib.sha256
       ).hexdigest()

   def verify_webhook_signature(
       payload: dict,
       signature: str,
       secret: str
   ) -> bool:
       expected = sign_webhook(payload, secret)
       return hmac.compare_digest(signature, expected)
   ```

4. 创建 `src/saw/webhooks/queue.py`：
   ```python
   import asyncio
   import httpx
   from datetime import datetime

   class WebhookQueue:
       def __init__(self, redis: RedisCache):
           self.redis = redis
           self.client = httpx.AsyncClient(timeout=10.0)

       async def enqueue(self, webhook_id: str, event: WebhookEvent):
           # 添加到 Redis 队列
           await self.redis.client.lpush(
               "webhook_queue",
               json.dumps({
                   "webhook_id": webhook_id,
                   "event": event.dict()
               })
           )

       async def process_queue(self):
           while True:
               # 从队列获取任务
               task = await self.redis.client.brpop("webhook_queue", timeout=5)
               if task:
                   await self._deliver_webhook(json.loads(task[1]))

       async def _deliver_webhook(self, task: dict):
           webhook_id = task["webhook_id"]
           event = WebhookEvent(**task["event"])

           # 获取 Webhook 配置
           webhook = await self.session.get(Webhook, webhook_id)
           if not webhook or not webhook.is_active:
               return

           # 签名
           signature = sign_webhook(event.dict(), webhook.secret)

           # 发送请求
           for attempt in range(3):
               try:
                   response = await self.client.post(
                       webhook.url,
                       json=event.dict(),
                       headers={
                           "Content-Type": "application/json",
                           "X-SAW-Signature": signature,
                           "X-SAW-Event": event.event_type,
                       }
                   )
                   if response.status_code < 400:
                       webhook.last_triggered_at = datetime.utcnow()
                       webhook.failure_count = 0
                       await self.session.commit()
                       return
               except Exception as e:
                   pass

               await asyncio.sleep(5 * (attempt + 1))

           # 所有重试失败
           webhook.failure_count += 1
           if webhook.failure_count >= 10:
               webhook.is_active = False
           await self.session.commit()
   ```
