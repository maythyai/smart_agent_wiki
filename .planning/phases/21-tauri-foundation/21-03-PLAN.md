       ```rust
       use tauri::{AppHandle, Manager, WebviewWindow};

       #[tauri::command]
       pub async fn get_system_theme(window: WebviewWindow) -> Result<String, String> {
           let theme = window.theme()
               .map_err(|e| e.to_string())?;

           let theme_str = match theme {
               tauri::Theme::Dark => "dark",
               tauri::Theme::Light => "light",
               _ => "light",
           };

           Ok(theme_str.to_string())
       }

       pub fn setup_theme_listener(app: &AppHandle) -> Result<(), Box<dyn std::error::Error>> {
           // Emit initial theme
           if let Some(window) = app.get_webview_window("main") {
               if let Ok(theme) = window.theme() {
                   let is_dark = matches!(theme, tauri::Theme::Dark);
                   let _ = window.emit("theme-changed", is_dark);
               }
           }

           Ok(())
       }
       ```

       ```typescript
       import { useEffect, useState } from 'react';
       import { invoke } from '@tauri-apps/api/core';
       import { listen } from '@tauri-apps/api/event';

       export function useTheme() {
         const [isDark, setIsDark] = useState(false);
         const [isReady, setIsReady] = useState(false);

         useEffect(() => {
           // Check if running in Tauri
           const inTauri = typeof window !== 'undefined' && '__TAURI__' in window;
           if (!inTauri) {
             // Fall back to system preference via CSS
             const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
             setIsDark(mediaQuery.matches);
             setIsReady(true);

             const handler = (e: MediaQueryListEvent) => setIsDark(e.matches);
             mediaQuery.addEventListener('change', handler);
             return () => mediaQuery.removeEventListener('change', handler);
           }

           // Tauri mode
           const init = async () => {
             try {
               const theme = await invoke<string>('get_system_theme');
               setIsDark(theme === 'dark');
               setIsReady(true);
             } catch {
               setIsReady(true);
             }
           };

           init();

           const unlisten = listen<boolean>('theme-changed', (event) => {
             setIsDark(event.payload);
           });

           return () => {
             unlisten.then(fn => fn());
           };
         }, []);

         useEffect(() => {
           if (isReady) {
             document.documentElement.classList.toggle('dark', isDark);
           }
         }, [isDark, isReady]);

         return { isDark, isReady };
       }
       ```

       ```typescript
       import { useEffect } from 'react';
       import { listen } from '@tauri-apps/api/event';

       export function useShortcuts(handlers: Record<string, () => void>) {
         useEffect(() => {
           const inTauri = typeof window !== 'undefined' && '__TAURI__' in window;
           if (!inTauri) return;

           const unsubscribers: Promise<() => void>[] = [];

           Object.entries(handlers).forEach(([shortcut, handler]) => {
             unsubscribers.push(
               listen(`shortcut:${shortcut}`, () => handler())
             );
           });

           return () => {
             unsubscribers.forEach(p => p.then(fn => fn()));
           };
         }, [handlers]);
       }
       ```
