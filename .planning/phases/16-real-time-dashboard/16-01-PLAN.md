       ```python
       from fastapi import WebSocket
       from typing import Dict, Set
       import asyncio
       import json
       from datetime import datetime
       
       class ConnectionManager:
           def __init__(self):
               self.active_connections: Dict[str, WebSocket] = {}
               self.subscriptions: Dict[str, Set[str]] = {}  # platform -> client_ids
               self._heartbeat_task = None
           
           async def connect(self, websocket: WebSocket, client_id: str):
               await websocket.accept()
               self.active_connections[client_id] = websocket
               # Start heartbeat if first connection
               if len(self.active_connections) == 1:
                   self._heartbeat_task = asyncio.create_task(self._heartbeat_loop())
           
           def disconnect(self, client_id: str):
               self.active_connections.pop(client_id, None)
               # Remove from all subscriptions
               for platform_subs in self.subscriptions.values():
                   platform_subs.discard(client_id)
               # Stop heartbeat if no connections
               if not self.active_connections and self._heartbeat_task:
                   self._heartbeat_task.cancel()
           
           def subscribe(self, platform: str, client_id: str):
               if platform not in self.subscriptions:
                   self.subscriptions[platform] = set()
               self.subscriptions[platform].add(client_id)
           
           def unsubscribe(self, platform: str, client_id: str):
               if platform in self.subscriptions:
                   self.subscriptions[platform].discard(client_id)
           
           async def broadcast(self, message: dict):
               dead_clients = []
               for client_id, ws in self.active_connections.items():
                   try:
                       await ws.send_json(message)
                   except Exception:
                       dead_clients.append(client_id)
               for client_id in dead_clients:
                   self.disconnect(client_id)
           
           async def broadcast_to_subscribers(self, platform: str, message: dict):
               if platform not in self.subscriptions:
                   return
               dead_clients = []
               for client_id in self.subscriptions[platform]:
                   if client_id in self.active_connections:
                       try:
                           await self.active_connections[client_id].send_json(message)
                       except Exception:
                           dead_clients.append(client_id)
               for client_id in dead_clients:
                   self.disconnect(client_id)
           
           async def _heartbeat_loop(self):
               while self.active_connections:
                   await asyncio.sleep(30)
                   message = {"type": "ping", "timestamp": datetime.utcnow().isoformat()}
                   await self.broadcast(message)
       
       # Global instance
       manager = ConnectionManager()
       ```

       from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Depends
       from .websocket import manager
       from ..connectors.health_monitor import HealthMonitor, HealthStatus
       from ..connectors.sync_status import SyncStatusTracker
       import json
       import uuid
       
       router = APIRouter()
       
       @router.websocket("/integrations")
       async def integrations_websocket(websocket: WebSocket):
           client_id = str(uuid.uuid4())[:8]
           await manager.connect(websocket, client_id)
           
           try:
               # Send initial connection status
               await websocket.send_json({
                   "type": "connection_status",
                   "data": {
                       "connected": True,
                       "client_id": client_id,
                       "server_time": datetime.utcnow().isoformat()
                   }
               })
               
               while True:
                   data = await websocket.receive_text()
                   try:
                       msg = json.loads(data)
                       if msg.get("action") == "subscribe" and "platform" in msg:
                           manager.subscribe(msg["platform"], client_id)
                           await websocket.send_json({
                               "type": "subscribed",
                               "platform": msg["platform"]
                           })
                       elif msg.get("action") == "unsubscribe" and "platform" in msg:
                           manager.unsubscribe(msg["platform"], client_id)
                           await websocket.send_json({
                               "type": "unsubscribed",
                               "platform": msg["platform"]
                           })
                   except json.JSONDecodeError:
                       await websocket.send_json({
                           "type": "error",
                           "message": "Invalid JSON"
                       })
           except WebSocketDisconnect:
               manager.disconnect(client_id)
       
       # Integration with HealthMonitor
       async def broadcast_health_change(platform: str, health: ConnectorHealth):
           await manager.broadcast_to_subscribers(platform, {
               "type": "connector_health",
               "platform": platform,
               "data": {
                   "status": health.status.value,
                   "last_success_at": health.last_success_at.isoformat() if health.last_success_at else None,
                   "last_failure_at": health.last_failure_at.isoformat() if health.last_failure_at else None,
                   "consecutive_failures": health.consecutive_failures,
                   "last_error": health.last_error
               }
           })
       
       async def broadcast_sync_progress(platform: str, status: SyncStatus):
           await manager.broadcast_to_subscribers(platform, {
               "type": "sync_progress",

                   "items_synced": status.items_synced,

                   "completion_percent": status.completion_percent,
                   "last_error": status.last_error
               }
           })
