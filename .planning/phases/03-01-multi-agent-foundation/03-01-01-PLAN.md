    # src/engines/collaborate/dispatcher.py
    from enum import Enum
    from typing import Literal
    import asyncio

    class ModelTier(Enum):
        HAIKU = "haiku"
        SONNET = "sonnet"
        OPUS = "opus"
        RULE = "rule"  # Guardian, no LLM

    MODEL_MAPPING = {
        ModelTier.HAIKU: "claude-3-5-haiku-latest",
        ModelTier.SONNET: "claude-sonnet-4-20250514",
        ModelTier.OPUS: "claude-opus-4-20250514",
    }

    FALLBACK_ORDER = {
        ModelTier.OPUS: [ModelTier.SONNET, ModelTier.HAIKU],
        ModelTier.SONNET: [ModelTier.HAIKU],
        ModelTier.HAIKU: [],  # 无降级
    }

    class AgentDispatcher:
        """Agent 调度器，负责模型路由和降级。"""

        def __init__(
            self,
            llm_router: LLMRouter,
            agents: dict[str, AgentProtocol],
            config: DispatcherConfig | None = None,
        ):
            self._llm = llm_router
            self._agents = agents
            self._config = config or DispatcherConfig()

        async def dispatch(
            self,
            agent_name: str,
            task: AgentTask,
            context: AgentContext,
            tools: list[Tool] | None = None,
        ) -> AgentResult:
            """调度 Agent 执行任务，带模型降级支持。"""
            agent = self._agents.get(agent_name)
            if not agent:
                raise AgentNotFoundError(f"Agent '{agent_name}' not found")

            # Guardian 等 rule-based agent 不需要模型
            if agent.model_tier == "rule":
                return await agent.execute(task, context, tools or [])

            # 尝试主模型，失败则降级
            tier = ModelTier(agent.model_tier)
            for fallback_tier in [tier] + FALLBACK_ORDER.get(tier, []):
                try:
                    # 设置 allowed_fails=3 防止激进冷却
                    result = await agent.execute(task, context, tools or [])
                    result.metadata["model_tier_used"] = fallback_tier.value
                    return result
                except RateLimitError:
                    if fallback_tier == FALLBACK_ORDER.get(tier, [])[-1] if FALLBACK_ORDER.get(tier) else fallback_tier:
                        raise  # 最后一个降级选项也失败
                    continue  # 尝试下一个降级
                except Exception:
                    raise

            raise DispatchError(f"All model tiers exhausted for agent {agent_name}")

        def get_model_for_tier(self, tier: ModelTier) -> str:
            """返回 tier 对应的实际模型名称。"""
            return MODEL_MAPPING.get(tier, MODEL_MAPPING[ModelTier.SONNET])

    @dataclass
    class DispatcherConfig:
        allowed_fails: int = 3  # PITFALLS.md 推荐
        cooldown_time: int = 120  # 秒
        timeout: int = 60  # 单次调用超时
    ```

    关键实现点（per PITFALLS.md Pitfall 2）：

    2. cooldown_time=120 秒
    3. 支持独立的 API key per tier（通过 LLMRouter 配置）
    4. 降级顺序：Opus → Sonnet → Haiku

    # src/engines/collaborate/a2a_protocol.py
    from dataclasses import dataclass, field
    from enum import Enum
    from datetime import datetime
    import uuid
    import asyncio

    A2A_PROTOCOL_VERSION = "1.0.0"

    class MessageType(Enum):
        REQUEST = "request"      # 请求另一个 Agent 执行操作
        RESPONSE = "response"    # 对请求的响应
        BROADCAST = "broadcast"  # 单向通知
        QUERY = "query"          # 查询信息
        RESULT = "result"        # 查询结果
        HANDOFF = "handoff"      # 任务移交（D-17）

    @dataclass
    class A2AMessage:
        """Agent-to-Agent 消息格式 (per D-15)。"""
        message_id: str = field(default_factory=lambda: str(uuid.uuid4()))
        sender: str = ""
        receiver: str | list[str] = ""  # 单 Agent 或广播
        message_type: MessageType = MessageType.REQUEST
        action: str = ""
        payload: dict = field(default_factory=dict)
        context: dict = field(default_factory=dict)  # workflow_id, step 等
        trace_id: str = field(default_factory=lambda: str(uuid.uuid4()))
        correlation_id: str | None = None  # request/response 配对
        timestamp: datetime = field(default_factory=datetime.utcnow)
        protocol_version: str = A2A_PROTOCOL_VERSION

        def to_dict(self) -> dict:
            return {
                "message_id": self.message_id,
                "sender": self.sender,
                "receiver": self.receiver,
                "message_type": self.message_type.value,
                "action": self.action,
                "payload": self.payload,
                "context": self.context,
                "trace_id": self.trace_id,
                "correlation_id": self.correlation_id,
                "timestamp": self.timestamp.isoformat(),
                "protocol_version": self.protocol_version,
            }

    @dataclass
    class A2AResult:
        """A2A 消息投递结果。"""
        success: bool
        message_id: str
        response: A2AMessage | None = None
        error: str | None = None

    class A2AAdapter:
        """A2A 协议适配器，处理消息路由和签名。"""

        def __init__(
            self,
            agents: dict[str, AgentProtocol],
            audit_signer: AuditSigner,  # Phase 02 复用
            dispatcher: AgentDispatcher,
        ):
            self._agents = agents
            self._signer = audit_signer
            self._dispatcher = dispatcher
            self._message_queue: asyncio.Queue[A2AMessage] = asyncio.Queue()

        def validate_version(self, message: A2AMessage) -> bool:
            """版本协商 (Pitfall 15)。"""
            # 支持语义化版本：主版本必须匹配
            msg_major = message.protocol_version.split(".")[0]
            our_major = A2A_PROTOCOL_VERSION.split(".")[0]
            return msg_major == our_major

        async def send(self, message: A2AMessage) -> A2AResult:
            """发送 A2A 消息。"""
            # 1. 版本检查
            if not self.validate_version(message):
                return A2AResult(
                    success=False,
                    message_id=message.message_id,
                    error=f"Incompatible protocol version: {message.protocol_version}",
                )

            # 2. 生成签名收据 (D-18)
            receipt = self._signer.sign_receipt(
                operation="a2a_message",
                data=message.to_dict(),
            )

            # 3. 路由消息
            if isinstance(message.receiver, list):
                # 广播模式
                results = await asyncio.gather(*[
                    self._deliver_to_agent(agent, message)
                    for agent in message.receiver
                ], return_exceptions=True)
                return A2AResult(success=True, message_id=message.message_id)
            else:
                # 直接消息
                return await self._deliver_to_agent(message.receiver, message)

        async def _deliver_to_agent(
            self,
            agent_name: str,
            message: A2AMessage,
        ) -> A2AResult:
            """投递消息到指定 Agent。"""
            agent = self._agents.get(agent_name)
            if not agent:
                return A2AResult(
                    success=False,
                    message_id=message.message_id,
                    error=f"Agent {agent_name} not found",
                )

            # 转换消息为 Agent 任务
            task = AgentTask(
                type=message.action,
                payload=message.payload,
                correlation_id=message.correlation_id,
            )
            context = AgentContext(
                wiki_state={},
                claims_context=[],
                workflow_id=message.context.get("workflow_id"),
                calling_agent=message.sender,
            )

            try:
                result = await agent.execute(task, context, [])

                # 生成响应消息
                if message.message_type in [MessageType.REQUEST, MessageType.QUERY]:
                    return A2AResult(
                        success=True,
                        message_id=message.message_id,
                        response=A2AMessage(
                            sender=agent_name,
                            receiver=message.sender,
                            message_type=MessageType.RESPONSE if message.message_type == MessageType.REQUEST else MessageType.RESULT,
                            payload=result.payload,
                            correlation_id=message.message_id,
                        ),
                    )
                return A2AResult(success=True, message_id=message.message_id)
            except Exception as e:
                return A2AResult(
                    success=False,
                    message_id=message.message_id,
                    error=str(e),
                )

        async def broadcast(self, sender: str, payload: dict, receivers: list[str]) -> A2AResult:
            """广播消息到多个 Agent。"""
            message = A2AMessage(
                sender=sender,
                receiver=receivers,
                message_type=MessageType.BROADCAST,
                payload=payload,
            )
            return await self.send(message)

        async def handoff(
            self,
            sender: str,
            receiver: str,
            task_type: str,
            payload: dict,
            context: dict,
        ) -> A2AResult:
            """任务移交，包含完整上下文 (D-17)。"""
            message = A2AMessage(
                sender=sender,
                receiver=receiver,
                message_type=MessageType.HANDOFF,
                action=task_type,
                payload=payload,
                context=context,  # 完整上下文传递
            )
            return await self.send(message)
    ```

    3. 支持同步（REQUEST/RESPONSE）和异步（BROADCAST）模式
